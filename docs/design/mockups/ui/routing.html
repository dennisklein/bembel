# address group login-nodes, revision 7, edited by a.berger 2026-10-30
name: login-nodes
owner: hpc-ops
description: login and gateway hosts of Hall 2
filter:
  all:
    - host.role: login
    - vrf: global
    - team: hpc-ops
  any_of_tags:
    - environment: prod
include:
  addresses: primary        # the primary address of each host
  prefixes:  [ 198.51.100.0/28 ]  # announced login VIP range
families: [ 4, 6 ]
feeds:
  nftables: { set: login_nodes, table: inet filter }
  edl: { ttl: 300 }
  firewalld: { ipset: login-nodes }
